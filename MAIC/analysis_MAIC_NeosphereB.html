Bootstrap¶

In [15]:
def estimand_fn(w=w, y=y_input, idx=None):
    # multiplier bootstrap uses same rows -> idx is None
    yy = y if idx is None else y[idx]

    mean_y_counter = float(np.sum(w * yy))
    return {
        "mean_y_counter": mean_y_counter,
    }

boot = eb.bootstrap(
    B=1000,
    estimand_fn=estimand_fn,
    method="multiplier",
    seed=rng,
    warm_start=True,
)

print("\nBootstrap convergence rate:", boot["converged"].mean())
print("Bootstrap ESS (5%,50%,95%):", np.percentile(boot["ess"], [5, 50, 95]))
print("Bootstrap mean_y (2.5%,50%,97.5%):", np.percentile(boot["mean_y_counter"], [2.5, 50, 97.5]))
